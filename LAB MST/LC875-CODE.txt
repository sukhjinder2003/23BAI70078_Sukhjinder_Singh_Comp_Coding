class Solution {
public:

    bool check(vector<int>& piles, int h, int mid){
        long long hours = 0;

        for(int i = 0; i < piles.size(); i++){
            hours += piles[i] / mid;

            if(piles[i] % mid != 0){
                hours++;
            }

            if(hours > h){
                return false;
            }
        }

        return true;
    }

    int minEatingSpeed(vector<int>& piles, int h) {
        int l = 1;
        int r = *max_element(piles.begin(), piles.end());
        int res = r;

        while(l <= r){
            int mid = l+ (r-l)/2;

            if(check(piles, h, mid)){
                res = mid;
                r = mid - 1;
            }
            else{
                l = mid + 1;
            }
        }

        return res;
        
    }
};
